cmake_minimum_required(VERSION 3.20)
project(ModeratedLMS VERSION 1.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LMS_COVERAGE "Instrument first-party code" OFF)
find_path(SODIUM_INCLUDE_DIR sodium.h REQUIRED)
find_library(SODIUM_LIBRARY NAMES sodium libsodium REQUIRED)
find_package(Threads REQUIRED)
add_library(sqlite STATIC vendor/sqlite/sqlite-amalgamation-3450100/sqlite3.c)
target_include_directories(sqlite PUBLIC vendor/sqlite/sqlite-amalgamation-3450100)
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1)
add_library(lms_core src/domain.cpp src/repository.cpp src/services.cpp)
target_include_directories(lms_core PUBLIC include vendor ${SODIUM_INCLUDE_DIR})
target_link_libraries(lms_core PUBLIC sqlite ${SODIUM_LIBRARY} Threads::Threads)
target_compile_options(lms_core PRIVATE -Wall -Wextra -Wpedantic)
if(LMS_COVERAGE AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_compile_options(lms_core PUBLIC --coverage -O0 -g)
  target_link_options(lms_core PUBLIC --coverage)
endif()
add_executable(lms_server src/server.cpp)
target_link_libraries(lms_server PRIVATE lms_core)
add_executable(lms_client src/client.cpp)
target_include_directories(lms_client PRIVATE vendor)
target_link_libraries(lms_client PRIVATE Threads::Threads)
if(WIN32)
  enable_language(RC)
  add_executable(lms_gui WIN32 src/gui.cpp src/desktop.rc)
  target_include_directories(lms_gui PRIVATE include vendor)
  target_compile_definitions(lms_gui PRIVATE UNICODE _UNICODE)
  target_link_libraries(lms_gui PRIVATE Threads::Threads ws2_32 crypt32 comctl32 comdlg32 gdi32 user32)
  install(TARGETS lms_gui RUNTIME DESTINATION bin)
  target_link_libraries(lms_server PRIVATE ws2_32 crypt32)
  target_link_libraries(lms_client PRIVATE ws2_32 crypt32)
endif()
configure_file(database/schema.sql schema.sql COPYONLY)
configure_file(database/migration-v1.1.sql migration-v1.1.sql COPYONLY)
target_compile_definitions(lms_server PRIVATE LMS_MIGRATION_PATH="${CMAKE_CURRENT_BINARY_DIR}/migration-v1.1.sql")
include(CTest)
if(BUILD_TESTING)
  if(WIN32)
    add_executable(lms_gui_tests WIN32 src/gui.cpp src/desktop.rc)
    target_include_directories(lms_gui_tests PRIVATE include vendor)
    target_compile_definitions(lms_gui_tests PRIVATE UNICODE _UNICODE LMS_GUI_TEST)
    target_link_libraries(lms_gui_tests PRIVATE Threads::Threads ws2_32 crypt32 comctl32 comdlg32 gdi32 user32)
    add_test(NAME lms_gui_forms COMMAND lms_gui_tests)
    set_tests_properties(lms_gui_forms PROPERTIES WORKING_DIRECTORY ${CMAKE_BINARY_DIR})
  endif()
  add_executable(lms_tests tests/tests.cpp)
  target_link_libraries(lms_tests PRIVATE lms_core)
  add_test(NAME lms_tests COMMAND lms_tests)
  set_tests_properties(lms_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_BINARY_DIR})
endif()
install(TARGETS lms_server lms_client RUNTIME DESTINATION bin)
install(FILES database/schema.sql database/migration-v1.1.sql DESTINATION bin)
